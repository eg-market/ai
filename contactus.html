<!-- Message Modal -->
<div id="messageModal" style="display: none;">
<h2>Send Us a Message</h2>
<textarea id="userMessage" placeholder="Write your message here..." rows="5"></textarea>
<button id="sendMessageBtn">Send Message</button>
<button id="closeModalBtn">Close</button>
</div>

<script src="https://cdn.emailjs.com/dist/email.min.js"></script>
<script>
// Initialize EmailJS
(function() {
emailjs.init("LgUkZLzt8N_NS"); // Replace with your EmailJS public key
})();

// Show the modal when the button is clicked
document.getElementById('contactUsBtn').onclick = function() {
document.getElementById('messageModal').style.display = 'block';
};

// Close the modal
document.getElementById('closeModalBtn').onclick = function() {
document.getElementById('messageModal').style.display = 'none';
};

// Send the message
document.getElementById('sendMessageBtn').onclick = function() {
const message = document.getElementById('userMessage').value;

const templateParams = {
to_email: 'wlid11@yahoo.com', // Your email address
message: message,
};

// Send the email
emailjs.send('YOUR_SERVICE_ID', 'YOUR_TEMPLATE_ID', templateParams)
.then(function(response) {
alert('Message sent successfully!');
document.getElementById('userMessage').value = ''; // Clear the textarea
document.getElementById('messageModal').style.display = 'none'; // Close the modal
}, function(error) {
alert('Failed to send message. Please try again.');
});
};
</script>
